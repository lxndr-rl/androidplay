cmake_minimum_required(VERSION 3.10)
project(CarPlayService CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenSSL REQUIRED)
add_library(iap2link src/iap2_link.cpp src/iap2_control.cpp src/iap2_session.cpp src/iap2_driver.cpp src/bplist.cpp src/airplay.cpp src/airplay_client.cpp)
target_link_libraries(iap2link PUBLIC OpenSSL::Crypto)
target_include_directories(iap2link PUBLIC include)
enable_testing()
add_executable(test_iap2_link tests/test_iap2_link.cpp)
target_link_libraries(test_iap2_link iap2link)
add_test(NAME iap2_link COMMAND test_iap2_link)
add_executable(test_iap2_session tests/test_iap2_session.cpp)
target_link_libraries(test_iap2_session iap2link)
add_test(NAME iap2_session COMMAND test_iap2_session)
add_executable(test_iap2_driver tests/test_iap2_driver.cpp)
target_link_libraries(test_iap2_driver iap2link)
add_test(NAME iap2_driver COMMAND test_iap2_driver)
add_executable(test_airplay tests/test_airplay.cpp)
target_link_libraries(test_airplay iap2link)
add_test(NAME airplay COMMAND test_airplay)
add_executable(test_airplay_client tests/test_airplay_client.cpp)
target_link_libraries(test_airplay_client iap2link)
add_test(NAME airplay_client COMMAND test_airplay_client)
